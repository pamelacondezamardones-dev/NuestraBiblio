
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Usuarios autorizados de NuestraBiblio
    function autorizado() {
      return request.auth != null &&
        request.auth.uid in [
          'tISgQLVocgaWYqEPzCJW3FwNg4n1',
          'jje69QZwcEdkZ0YuQj7fW2fXLto2'
        ];
    }

    function propietario() {
      return autorizado() &&
        resource.data.uid == request.auth.uid;
    }

    function nuevoPropietario() {
      return autorizado() &&
        request.resource.data.uid == request.auth.uid;
    }

    function conservaPropietario() {
      return request.resource.data.uid ==
        resource.data.uid;
    }

    // 1. Catálogo familiar
    match /libros/{libroId} {
      allow read, create, update, delete:
        if autorizado();
    }

    // 2. Lecturas personales
    match /lecturas/{lecturaId} {
      allow read, delete: if propietario();

      allow create: if nuevoPropietario();

      allow update: if propietario() &&
        conservaPropietario();
    }

    // 3. Frases personales
    match /frases/{fraseId} {
      allow read, delete: if propietario();

      allow create: if nuevoPropietario();

      allow update: if propietario() &&
        conservaPropietario();
    }

    // 4. Opiniones compartidas
    match /opiniones/{opinionId} {
      allow read: if autorizado();

      allow create: if nuevoPropietario();

      allow update: if propietario() &&
        conservaPropietario();

      allow delete: if propietario();
    }

    // 5. Préstamos compartidos
    match /prestamos/{prestamoId} {
      allow read, create, update, delete:
        if autorizado();
    }

    // 6. Lista familiar de deseos
    match /deseos/{deseoId} {
      allow read, create, update, delete:
        if autorizado();
    }

    // 7. Metas familiares
    match /metas/{metaId} {
      allow read: if autorizado();

      allow create: if nuevoPropietario();

      allow update: if propietario() &&
        conservaPropietario();

      allow delete: if propietario();
    }

    // Todo lo demás queda denegado.
  }
}
